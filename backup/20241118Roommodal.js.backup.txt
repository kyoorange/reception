// Fetchを使用したAPIリクエスト例
async function fetchOccupies() {
    const response = await fetch('/api/v1/occupies');
    const data = await response.json();
    console.log(data); // 取得したデータを確認
}

// 新しいOccupyを作成する例
async function createOccupy(timeLength, returnedAt) {
    const response = await fetch('/api/v1/occupies', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json'
        },
        body: JSON.stringify({ occupy: { time_length: timeLength, returned_at: returnedAt } })
    });
    const data = await response.json();
    console.log(data); // 作成したデータを確認
}

async function fetchRegistrants() {
